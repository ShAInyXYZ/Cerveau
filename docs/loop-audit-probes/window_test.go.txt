package window

import (
	"context"
	"strings"
	"testing"
	"cerveau/internal/llm"
)

func auditCount(_ context.Context, text string) int { return len(text) }
func auditTotal(msgs []llm.Message) int {
	n := 0
	for _, msg := range msgs {
		n += len(msg.Content)+4
		for _, tc := range msg.ToolCalls { n += len(tc.Function.Arguments)+len(tc.Function.Name) }
	}
	return n
}

func TestAuditL15DemotionAndTrimAccounting(t *testing.T) {
	m := NewManager(1000, 100, CounterFunc(auditCount))
	items := []Item{{Kind:"system", Msg:llm.Message{Role:"system", Content:"system"}},
		{Kind:"tool", EvtID:"evt_1", Msg:llm.Message{Role:"tool", Content:strings.Repeat("x", 2000)}}}
	for i:=0; i<6; i++ { items=append(items, Item{Kind:"user", Msg:llm.Message{Role:"user", Content:strings.Repeat("x", 200)}}) }
	msgs, rep := m.Build(context.Background(), items)
	if actual:=auditTotal(msgs); rep.Tokens != actual { t.Fatalf("reported=%d, actual=%d, demoted=%d trimmed=%d", rep.Tokens, actual, rep.Demoted, rep.Trimmed) }
}

func TestAuditL15CompactionPreservesToolPairs(t *testing.T) {
	m := NewManager(1000, 100, CounterFunc(auditCount))
	items := []Item{{Kind:"system", Msg:llm.Message{Role:"system", Content:"system"}},
		{Kind:"assistant", Msg:llm.Message{Role:"assistant", Content:strings.Repeat("x", 1000), ToolCalls:[]llm.ToolCall{{ID:"call1"}}}},
		{Kind:"tool", Msg:llm.Message{Role:"tool", ToolCallID:"call1", Content:"result"}}}
	for i:=0; i<5; i++ { items=append(items, Item{Kind:"user", Msg:llm.Message{Role:"user", Content:"next"}}) }
	msgs, _ := m.Build(context.Background(), items)
	seen:=map[string]bool{}
	for _, msg:=range msgs {
		for _, tc:=range msg.ToolCalls { seen[tc.ID]=true }
		if msg.Role=="tool" && !seen[msg.ToolCallID] { t.Fatalf("orphaned tool result: %s", msg.ToolCallID) }
	}
}
